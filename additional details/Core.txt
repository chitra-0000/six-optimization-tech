'Public procedure to flush SIX data from the database
'Optional param allows to pinpoint the data from which SIX file should be removed. If omitted, all SIX data is affected
Public Sub FlushSixIsin(Optional ByVal iSourceFileId As Variant)

    Dim sSQLDelete As String

    'Defining the common part of the DELETE SQL query
    sSQLDelete = "DELETE FROM SIX_ISIN"

    'If an optional param is provided...
    If Not IsMissing(iSourceFileId) Then

        '... then the SQL query is amended accordingly to pinpont the data to be deleted
        sSQLDelete = sSQLDelete & " WHERE rsf_id = " & CStr(iSourceFileId)

    End If

    'Execution of the obtained query
    CurrentDb.Execute sSQLDelete

End Sub

'Public procedure to remove the information of each source file that has been loaded (tracking system/CURRENTLY_LOADED_FILES table)
'Mandatory param is an ID referenced in table REF_SOURCE_FILE, to pinpoint which SIX file is to be removed.
Public Sub FlushSixLoadedFile(ByVal iSourceFileId As Integer)

    Dim sSixFileRefLabel As String
    Dim rSixFileRef As Recordset

    'Getting the file "ref_label" for the param-given ID. ref_label is used for the columns in table CURRENTLY_LOADED_FILES (see below)
    Set rSixFileRef = CurrentDb.OpenRecordset("SELECT * FROM REF_SOURCE_FILE WHERE rsf_id = " & iSourceFileId)

    sSixFileRefLabel = rSixFileRef.Fields("rsf_ref_label").Value

    rSixFileRef.Close

    Set rSixFileRef = Nothing

    'Setting and executing the SQL Update query, using the previously obtained "ref_label" to reach the proper columns in CURRENTLY_LOADED_FILES table
    CurrentDb.Execute "UPDATE CURRENTLY_LOADED_FILES" & vbCrLf & _
                      "SET " & Replace(LCase(sSixFileRefLabel), " ", "_") & "_file = NULL, " & Replace(LCase(sSixFileRefLabel), " ", "_") & "_loading_date = NULL" & vbCrLf & _
                      "WHERE ID = 1"

End Sub

'Public procedure to ultimately remove the unwanted Instrument rows from the database
'Called only when Exporting the data to XML (Form_Dashboard --> btnExportToXML_Click), to ensure the XML file do not contain those rows
Public Sub RemoveUnwantedInstruments()

    'Executing 4 small DELETE SQL queries, with different WHERE clauses
    CurrentDb.Execute "DELETE FROM SIX_ISIN WHERE rsf_id = 0 AND isin LIKE '" & sInstruNullIsinPattern & "*'"
    CurrentDb.Execute "DELETE FROM SIX_ISIN WHERE rsf_id = 0 AND target = 'CMIC' AND confidence_level < 25"
    CurrentDb.Execute "DELETE FROM SIX_ISIN WHERE rsf_id = 0 AND target <> 'CMIC' AND (sanctioned = 'No' OR confidence_level <= 25)"

    'CurrentDb.Execute "DELETE FROM SIX_ISIN WHERE rsf_id = 0 AND active_flag = 'Inactive'"

End Sub

'Public procedure to fill each ISIN with its "regime's" program in the dedicated table
Public Sub FillIsinPrograms()

    Dim sEUUSSelect, sUKSelect, sUSPPSelect, sSqlQuery As String

    'If the destination table is already filled...
    If CurrentDb.OpenRecordset("SELECT COUNT(*) AS CountRows FROM ISIN_PROGRAMS")!CountRows > 0 Then

        '... flush it
        FlushIsinPrograms

    End If

    'if the USPP Split checkbox is ticked...
    If Form_Dashboard.chkSplitUSPP.Value Then

        '... get the SQL filter text to get USPP rows differenciated from EU/US ISIN
        sUSPPSelect = GetUSPPISINSqlFilter

    Else '... if USPP checkbox not ticked...

        '... put "False" as SQL filter (will result in USPP rows not being segregated from EU/US ISIN)
        sUSPPSelect = "False"

    End If

    'Set SQL filters for EU/US, UK and USPP, depending on the USPP Split filter result
    sEUUSSelect = "(SUM(IIF(eu IS NOT NULL OR us_sdn IS NOT NULL OR us_non_sdn IS NOT NULL AND NOT " & sUSPPSelect & ", 1, 0)) > 0)"
    sUKSelect = "(SUM(IIF(uk IS NOT NULL, 1, 0)) > 0)"
    sUSPPSelect = "(SUM(IIF(" & sUSPPSelect & ", 1, 0)) > 0 AND NOT " & sEUUSSelect & ")"

    'Building SQL query using obtained filters to INSERT into destination tables
    sSqlQuery = "INSERT INTO ISIN_PROGRAMS (isin, is_eu_us, is_uk, is_uspp)" & vbCrLf & _
                "SELECT isin," & vbCrLf & vbTab & _
                        sEUUSSelect & "," & vbCrLf & vbTab & _
                        sUKSelect & "," & vbCrLf & vbTab & _
                        sUSPPSelect & vbCrLf & _
                        "FROM SIX_ISIN" & vbCrLf & _
                "GROUP BY ISIN"

    'Executing SQL query
    CurrentDb.Execute sSqlQuery

End Sub

'Public procedure to flush the destination table for ISINs "regime's" program
Public Sub FlushIsinPrograms()

    'Executing SQL DELETE query
    CurrentDb.Execute "DELETE FROM ISIN_PROGRAMS"

End Sub

    Set fdp = Nothing

'Public procedure to flush - partially or totally - the ISIN Historical Application Dates table
Public Sub FlushISINHistoApplicationDate(Optional ByVal iIdReglissFileOrigin As Integer = -1)

    'Executing the DELETE SQL query to flush the table. The WHERE clause depends on the procedure's parameter
    CurrentDb.Execute "DELETE FROM ISIN_HISTO_APPLICATION_DATE" & IIf(iIdReglissFileOrigin >= 0, " WHERE id_regliss_file_origin = " & CStr(iIdReglissFileOrigin), vbNullString)

End Sub

'Public procedure to reset the USPP filter to its original values (in the proper table)
Public Sub ResetToOriginalUSPPFilter()

    'Executing the DELETE SQL query to completely flush the table containing the USPP filter
    CurrentDb.Execute "DELETE FROM USPP_ISIN_SELECT_PARAMETERS"

    'Executing the STORED PROC putting back the original USPP filter
    CurrentDb.Execute "INIT_USPP_ISIN_SELECT_PARAMETERS"

End Sub

    Dim xmlElProgram As IXMLDOMElement

    ExistProgramNode = False

    If xmlElProgramsNode.HasChildNodes Then

        For Each xmlElProgram In xmlElProgramsNode.ChildNodes

            If xmlElProgram.getElementsByTagName("label").Item(0).nodeTypedValue = sProgramType & " - " & sProgramLabel And xmlElProgram.getElementsByTagName("description").Item(0).nodeTypedValue = sTarget Then

                ExistProgramNode = True
                Exit For

            End If
        Next
    End If
End Function

' Returns the properly Access SQL formatted US PERSONS PROHIBITION filter (to be used in the select query for XML export)
Public Function GetUSPPISINSqlFilter() As String

    Dim sSqlQuery As String
    Dim bIsTargetSet, bIsUsNonSdnProgram1Set, bIsusNonSdnProgram2Set, bIsIsinSqlPatternSet, bIsManyRows, bIsEmpty As Boolean
    Dim rMyRecordSet As Recordset

    'Getting the all the USPP ISIN SELECT PARAMETERS (sub-filters, 1 sub-filter per row)
    Set rMyRecordSet = CurrentDb.OpenRecordset("SELECT TRIM(uisp.target) AS target, TRIM(uisp.us_non_sdn_program1) AS us_non_sdn_program1, TRIM(uisp.us_non_sdn_program2) AS us_non_sdn_program2, TRIM(uisp.isin_sql_pattern) AS isin_sql_pattern FROM USPP_ISIN_SELECT_PARAMETERS uisp")

    With rMyRecordSet

        If Not (.BOF And .EOF) Then

            .MoveLast
            bIsManyRows = (.RecordCount > 1)
            GetUSPPISINSqlFilter = IIf(bIsManyRows, "(", vbNullString)
            .MoveFirst

            Do While Not .EOF

                GetUSPPISINSqlFilter = GetUSPPISINSqlFilter & IIf(.AbsolutePosition > 0, " OR ", "") & "("

                bIsTargetSet = (.Fields("target").Value <> vbNullString)
                bIsUsNonSdnProgram1Set = (.Fields("us_non_sdn_program1").Value <> vbNullString)
                bIsusNonSdnProgram2Set = (.Fields("us_non_sdn_program2").Value <> vbNullString)
                bIsIsinSqlPatternSet = (.Fields("isin_sql_pattern").Value <> vbNullString)

                If bIsTargetSet Then

                    GetUSPPISINSqlFilter = GetUSPPISINSqlFilter & "target = '" & .Fields("target").Value & "'"

                End If

                If bIsUsNonSdnProgram1Set Or bIsusNonSdnProgram2Set Then

                    If bIsTargetSet Then

                        GetUSPPISINSqlFilter = GetUSPPISINSqlFilter & " AND "

                    End If

                    GetUSPPISINSqlFilter = GetUSPPISINSqlFilter & "us_non_sdn"

                    If bIsUsNonSdnProgram1Set And bIsusNonSdnProgram2Set Then

                        GetUSPPISINSqlFilter = GetUSPPISINSqlFilter & " IN ('" & .Fields("us_non_sdn_program1").Value & "', '" & .Fields("us_non_sdn_program2").Value & "')"

                    Else

                        GetUSPPISINSqlFilter = GetUSPPISINSqlFilter & " = '" & IIf(bIsUsNonSdnProgram1Set, .Fields("us_non_sdn_program1").Value, .Fields("us_non_sdn_program2").Value) & "'"

                    End If
                End If

                If bIsIsinSqlPatternSet Then

                If bIsTargetSet Or bIsUsNonSdnProgram1Set Or bIsusNonSdnProgram2Set Then

                    GetUSPPISINSqlFilter = GetUSPPISINSqlFilter & " AND "

                End If

                GetUSPPISINSqlFilter = GetUSPPISINSqlFilter & "isin " & .Fields("isin_sql_pattern").Value

            End If

            GetUSPPISINSqlFilter = GetUSPPISINSqlFilter & ")"

            .MoveNext

        Loop
    End If

    .Close

    End With

    GetUSPPISINSqlFilter = IIf(GetUSPPISINSqlFilter = vbNullString, "False", GetUSPPISINSqlFilter & IIf(bIsManyRows, ")", vbNullString))

    Set rMyRecordSet = Nothing

End Function

' Opens the US PERSONS PROHIBITION filter Help document (included in the database as an attachement) with Microsoft Word
' As of today, only 1 document is attached. This design is purely made for hypothetical evolutions with many docs to be stored
Public Sub openUSPPFilterHelpDoc()

    Dim oWordApp As Word.Application
    Dim oWordDoc As Word.Document
    Dim rsHelpDocsMain, rsHelpDocsChild As Recordset
    Dim sHelpDocFileName, sTmpSavedStatus As String
    Dim foHelpDocContent As Field
    Dim lWindowHandler As Long
    Const iHelpDocId As Integer = 1

    ' Get the document id 1 from DOCS table in the database
    Set rsHelpDocsMain = getDocsFromDB(iHelpDocId)

    ' Use the obtained recordset (with only 1 row)...
    With rsHelpDocsMain

        ' ... and check if there actually is a row or more ...
        If Not (.BOF And .EOF) Then

            ' ... trick VBA to get the RecordCount set to the number of rows the Recordset actually contains
            .MoveLast
            .MoveFirst

            ' ... and if there actually is only 1 row (which is what's expected)...
            If .RecordCount = 1 Then

                ' ... get the "doc_content" field, which is also a Recordset, since 1 field can contain many documents
                Set rsHelpDocsChild = .Fields("doc_content").Value

                ' ... check if that there actually is at least 1 row (= at least 1 document) in the "doc_content"...
                If Not (rsHelpDocsChild.BOF And rsHelpDocsChild.EOF) Then

                    ' ... trick VBA to get the RecordCount set to the number of rows the Recordset actually contains
                    rsHelpDocsChild.MoveLast
                    rsHelpDocsChild.MoveFirst

                    ' If the Recordset actually contains only 1 row (= only 1 document, which is what's expected)...
                    If rsHelpDocsChild.RecordCount = 1 Then

                        ' ... get the filename and file content
                        sHelpDocFileName = rsHelpDocsChild.Fields("FileName").Value
                        Set foHelpDocContent = rsHelpDocsChild.Fields("FileData")

                        ' Try and save the file into the user's TEMP folder and catch the status
                        sTmpSavedStatus = saveTmpFileFromField(sHelpDocFileName, foHelpDocContent)

                        ' If the file saving status is either "SAVE_OK or "ALREADY_OPENED"...
                        If isin(sTmpSavedStatus, "SAVE_OK,ALREADY_OPENED") Then

                            ' ...only if the status is "SAVE_OK"...
                        If sTmpSavedStatus = "SAVE_OK" Then

                            ' ... define a new Microsoft Word application object, open the saved file with it and make Word visible
                            Set oWordApp = CreateObject("Word.Application")
                            Set oWordDoc = oWordApp.Documents.Open(getEnvironTempPath & sHelpDocFileName)
                            oWordApp.Visible = True

                        End If

                        ' ... in both status cases, try and put Word application window on the foreground
                        lWindowHandler = FindWindow(vbNullString, sHelpDocFileName & " - Word")

                        If lWindowHandler > 0 Then

                            SetForegroundWindow lWindowHandler

                        End If
                    Else ' ... if status wasn't SAVE_OK or ALREADY_OPENED ...

                        ' ... show a proper alert message
                        MsgBox "Un souci a été rencontré avec le fichier d'aide." & vbCrLf & "Il est possible qu'il soit toujours ouvert en mémoire par une session cachée de Word." & vbCrLf & "Si c'est le cas, veuillez fermer cette session de Word et réessayer.", vbExclamation + vbOKOnly, "Erreur avec le fichier d'aide"

                    End If
                End If
            End If

            ' Close the "doc_content" Recordset
            rsHelpDocsChild.Close

        End If
    End If

    ' Close the DOCS Recordset
    .Close

    End With
End Sub

' Flushes all HDD-saved in-base attached documents
' As of today, only 1 document is attached. This design is purely made for hypothetical evolutions with many docs to be stored
Public Sub FlushTempDocsFromHDD()

    Dim rsTempDocsMain, rsTempDocsChild As Recordset
    Dim sFileNameFullPath As String

    Set rsTempDocsMain = getDocsFromDB

    With rsTempDocsMain

        If Not (.BOF And .EOF) Then

            .MoveLast
            .MoveFirst

            Do While Not .EOF

                Set rsTempDocsChild = .Fields("doc_content").Value

                If Not (rsTempDocsChild.BOF And rsTempDocsChild.EOF) Then

                    rsTempDocsChild.MoveLast
                    rsTempDocsChild.MoveFirst

                    Do While Not rsTempDocsChild.EOF

                        sFileNameFullPath = getEnvironTempPath & rsTempDocsChild.Fields("FileName").Value

                        If isFileExist(sFileNameFullPath) Then

                            removeExistingFile sFileNameFullPath

                        End If

                        rsTempDocsChild.MoveNext

                    Loop

                    rsTempDocsChild.Close

                End If

                .MoveNext

            Loop

            .Close

        End If
    End With
End Sub

'return a program node reordered by programs label
Private Function OrderProgramsNodes(ByVal oPrograms As IXMLDOMElement) As IXMLDOMElement

    'Dim programNode, programLabelEl, programTypeEl, programDescriptionEl As IXMLDOMElement
    Dim oNewprogramsNode As IXMLDOMElement
    Set oNewprogramsNode = xmlDoc.createElement("programs")

    Dim aLabelsLists As Variant
    Set aLabelsLists = CreateObject("System.Collections.Arraylist")

    'Listing Programs labels
    For Each Child In oPrograms.ChildNodes
        aLabelsLists.Add Child.ChildNodes.Item(0).Text
    Next

    'Sorting Programs labels
    QuickSort aLabelsLists, 0, (aLabelsLists.Count - 1)

    'adding list in new order in new temp object
    For Each l In aLabelsLists
        For Each Child In oPrograms.ChildNodes
            If Child.ChildNodes.Item(0).nodeTypedValue = l Then
                oNewprogramsNode.appendChild Child.CloneNode(True)
            End If
        Next
    Next

    'returning xmlelement with ordered programs
    Set OrderProgramsNodes = oNewprogramsNode


    Set aLabelsLists = Nothing
    Set oNewprogramsNode = Nothing

End Function

'Public procedure to import a REGLISS GENERIC CSV file for historical Application Dates
Public Sub AutoImportReglissFile(ByVal iIdReglissFileOrigin As Integer, ByVal sReglissFullPtah As String)

    Dim sReglissFileName, sReglissFileNameNoPath, sIsin, sReglissFileRefLabel, sReglissFileHeader As String
    Dim iReglissFileIsinCol, iReglissFileIssueDateCol, iCntRows, iCurrentRowNb, iCurrentTaux, iPreviousTaux As Long
    Dim vaRowsFromFile, vaCurrentRow, vaColumnsFromFile As Variant
    Dim dIsinApplicationDate As Date
    Dim tsFile As TextStream
    Dim rReglissFileRef As Recordset
    Dim asReglissLists(3) As String



    'Setting an array with various possibilities of ISIN regimes. Its index is linked to the lists checkboxes
    asReglissLists(0) = "CMIC"
    asReglissLists(1) = "EU/US ISIN"
    asReglissLists(2) = "UK ISIN"
    asReglissLists(3) = "US PERSONS PROHIBITION ISIN"

    'Getting the information of the REGLISS GENERIC CSV file (rsf_id = 3) stored in the REF_SOURCE_FILE parameters-table by executing the proper SQL query
    Set rReglissFileRef = CurrentDb.OpenRecordset("SELECT * FROM REF_SOURCE_FILE WHERE rsf_id = 3")

    'Putting the query results in properly typed variables + closing resultset
    With rReglissFileRef

        sReglissFileRefLabel = .Fields("rsf_ref_label").Value
        sReglissFileHeader = .Fields("rsf_header").Value
        iReglissFileIsinCol = .Fields("rsf_isin_column").Value
        iReglissFileIssueDateCol = .Fields("rsf_issue_date_column").Value

        .Close

    End With


    'Catching the picked filename + stripping it from its path
    sReglissFileName = sReglissFullPtah
    sReglissFileNameNoPath = GetFilenameFromPath(sReglissFileName)

    'Checking if the chosen file is already opened. If not...
    If Not IsFileOpen(sReglissFileName) Then
        timerStartValue = GetTickCount

        'Set progressbar percentage to 0
        Form_Dashboard.ProgressBar.Value = 0
        Form_Dashboard.Etiquette79.Caption = "0 %"
        DoEvents
        iCntRows = countFileLines(sReglissFileName)

        'Instanciating a filestream object
        Set tsFile = CreateObject("Scripting.FileSystemObject").OpenTextFile(sReglissFileName, 1, True)

        'Setting filestream object parameters with the picked file
        With tsFile

            'Looping through each file's rows...
            Do While Not .AtEndOfStream

                'Incrementing by 1 the currently treated row number and setting the new percentage of the whole file it represents
                iCurrentRowNb = .Line
                iCurrentTaux = CInt((iCurrentRowNb * 100) / iCntRows)

                'Reading current row (and automatically preparing for the next one to be read)
                vaCurrentRow = .ReadLine

                'If this currently is the header row (1st file's row)...
                If iCurrentRowNb = 1 Then

                    '... Comparing this header to the expected header pattern defined for this kind of file. If they don't match...
                    If vaCurrentRow <> sReglissFileHeader Then

                        '... Ok only/Warning dialog box with proper message + Exit Sub without doing anything else
                        MsgBox "Le fichier choisi (" & sReglissFileNameNoPath & ") n'est pas un fichier CSV " & sReglissFileRefLabel & " standard !", vbExclamation, "Mauvais format de fichier"
                        Exit Sub

                    Else '... else means the current file header matches the stored header pattern...

                        '... flushing the already imported ISIN Historical Application Dates for the specific current list to be loaded
                        FlushISINHistoApplicationDate iIdReglissFileOrigin

                    End If

                '... else if the current row is not the 1st row and it is not empty...
                ElseIf Trim(vaCurrentRow) <> vbNullString Then

                    '... we split the row in an array using the semi-colon as separator
                    vaColumnsFromFile = Split(RemoveInValueSemiColon(vaCurrentRow), ";")

                    'we get the ISIN code and the Record Application Date from the array, using the indexes that were parametered for this kind of file
                    sIsin = QuotesIfNotNull(Trim(vaColumnsFromFile(iReglissFileIsinCol)))
                    dIsinApplicationDate = FormatDateStringToProperDate(Nz(vaColumnsFromFile(iReglissFileIssueDateCol), vbNullString))

                    'If there is indeed an ISIN...
                    If sIsin <> vbNullString Then

                        'Executing the SQL query to insert that ISIN + its Application Date + the file id to trace its origin
                        CurrentDb.Execute "INSERT INTO ISIN_HISTO_APPLICATION_DATE (isin, application_date, id_regliss_file_origin) VALUES (" & sIsin & ", " & FormatDateToProperStringDateSQL(dIsinApplicationDate) & ", " & CStr(iIdReglissFileOrigin) & ")"

                    End If
                End If

                'Applying new progress percentage to progress bar and refreshing display before looping to the next row
                If (iCurrentRowNb Mod 10 = 0) Or (iCurrentTaux <> iPreviousTaux) Then
                    Form_Dashboard.ProgressBar.Value = iCurrentTaux
                    Form_Dashboard.Etiquette79.Caption = iCurrentTaux & " %"
                    DoEvents
                End If
                iPreviousTaux = iCurrentTaux
            Loop

            .Close

        End With

        timerStopValue = GetTickCount
        timerTimeValue = ((timerStopValue - timerStartValue) + Abs((timerStopValue < timerStartValue) * 86400)) / 1000
        Select Case timerTimeValue
            Case Is < 1
                mssg = Format(timerTimeValue, "0.000 \m\s")
            Case 1 To 59
                mssg = Format(Int(timerTimeValue), "0 \s") 'this one rounds down
                'mssg = mssg & Format(timerTimeValue\1, "0 \s") this one rounds it off up or down
            Case 60 To 3599
                mssg = Format(Int(timerTimeValue / 60), "0 \m\, ") & Format(timerTimeValue Mod 60, "0 \s")
            Case Is >= 3600
                mssg = Format(Int(timerTimeValue / 3600), "0 \h\, ") & Format(Int((timerTimeValue Mod 3600) / 60), "0 \m\, ") & Format(timerTimeValue Mod 60, "0 \s")
            Case Else
                'do nothing
        End Select

        'Displaying Execution ouput debug
        currentDateTime = Now
        Debug.Print currentDateTime & "--- Fichier CSV " & sReglissFileRefLabel & " a bien été importé." & vbCrLf & vbTab & sReglissFileNameNoPath & vbCrLf & vbTab & _ '[cut off at screen edge - verify]
        "Total Time : " & mssg


    End If

End Sub

'Actions taken when the Export to XML button in clicked
Public Sub AutoExportToXML(ByVal sOutputFolder As String)

    Dim sFileSaveDirectory, sFilesSavedList, sExportResult As String



    If (chkEUUS.Value And (chkEUUSCMIC.Value Or chkEUUSISIN.Value)) Or chkUK.Value Or chkUSPP.Value Then

        'If unwated Instruments have not yes been removed...
        If Not bIsUnwantedInstruRemoved Then

            '...calling the homemade procedure which ultimately removes the unwanted Instruments that were kept for proper integration of the Options/Structured data
            RemoveUnwantedInstruments
            bIsUnwantedInstruRemoved = True

        End If

        'Calling the filler for ISIN PROGRAMS table now that only the ISIN we want are present in the main table
        FillIsinPrograms


            '... save the path
            sFileSaveDirectory = sOutputFolder
            sFilesSavedList = vbNullString

            If chkEUUS.Value Then

                If chkEUUSCMIC.Value Then

                    '... Save the file (using the homemade ExportXML function) and catching filename
                    sExportResult = ExportXML(0, sFileSaveDirectory, 1)

                    If Not sExportResult Like "ERR_*" Then

                        sFilesSavedList = sFilesSavedList & vbCrLf & vbTab & "- " & sExportResult

                    End If
                End If

                If chkEUUSISIN.Value Then

                    sExportResult = ExportXML(1, sFileSaveDirectory, 1)

                    If Not sExportResult Like "ERR_*" Then

                        '... Save the file (using the homemade ExportXML function) and catching filename
                        sFilesSavedList = sFilesSavedList & vbCrLf & vbTab & "- " & sExportResult

                    End If
                End If

            End If

            If chkUK.Value Then

                sExportResult = ExportXML(1, sFileSaveDirectory, 2)

                If Not sExportResult Like "ERR_*" Then

                    '... Save the file (using the homemade ExportXML function) and catching filename
                    sFilesSavedList = sFilesSavedList & vbCrLf & vbTab & "- " & sExportResult

                End If
            End If

            If chkUSPP.Value Then

                sExportResult = ExportXML(1, sFileSaveDirectory, 3)

                If Not sExportResult Like "ERR_*" Then

                    '... Save the file (using the homemade ExportXML function) and catching filename
                    sFilesSavedList = sFilesSavedList & vbCrLf & vbTab & "- " & sExportResult

                End If
            End If

            If sFilesSavedList <> vbNullString Then

                'MsgBox "Le(s) fichier(s) XML :" & vbCrLf & sFilesSavedList & vbCrLf & vbCrLf & "a/ont bien été enregistré(s).", vbInformation + vbOKOnly, "Fichier(s)
                XML bien exporté(s)"
                currentDateTime = Now
                Debug.Print currentDateTime & "--- Le(s) fichier(s) XML :" & vbCrLf & sFilesSavedList & vbCrLf & vbCrLf & "a/ont bien été enregistré(s)."
            Else

                MsgBox "Aucun fichier XML n'a été enregistré.", vbExclamation + vbOKOnly, "Aucun fichier XML exporté"

            End If
        Else 'if no folder was selected...

            '... display a warning message accordingly
            MsgBox "Aucun répertoire choisi. Aucun fichier XML n'a été exporté.", vbExclamation + vbOKOnly, "Pas de répertoire choisi"

        End If


End Sub
