'Actions taken when the Export to XML button in clicked
Private Sub btnExportToXML_Click()

    Dim sFileSaveDirectory, sFilesSavedList, sExportResult As String

    Block_Buttons

    If (chkEUUS.Value And (chkEUUSCMIC.Value Or chkEUUSISIN.Value)) Or chkUK.Value Or chkUSPP.Value Then

        'If unwated Instruments have not yes been removed...
        If Not bIsUnwantedInstruRemoved Then

            '...calling the homemade procedure which ultimately removes the unwanted Instruments that were kept for proper integration of the Options/Structured data
            RemoveUnwantedInstruments
            bIsUnwantedInstruRemoved = True

        End If

        'Calling the filler for ISIN PROGRAMS table now that only the ISIN we want are present in the main table
        FillIsinPrograms

        'Preparing a "Select Folder to Save file" dialog box...
        With Application.FileDialog(msoFileDialogFolderPicker)

            .AllowMultiSelect = False
            .Title = "Choisir le répertoire de sauvegarde des fichiers XML SIX"

            '... showing the dialog box and if a folder has been selected
            If .Show Then

                '... save the path
                sFileSaveDirectory = .SelectedItems(1)
                sFilesSavedList = vbNullString

                If chkEUUS.Value Then

                    If chkEUUSCMIC.Value Then

                        '... Save the file (using the homemade ExportXML function) and catching filename
                        sExportResult = ExportXML(0, sFileSaveDirectory, 1)

                        If Not sExportResult Like "ERR_*" Then

                            sFilesSavedList = sFilesSavedList & vbCrLf & vbTab & "- " & sExportResult

                        End If
                    End If

                    If chkEUUSISIN.Value Then

                        sExportResult = ExportXML(1, sFileSaveDirectory, 1)

                        If Not sExportResult Like "ERR_*" Then

                            '... Save the file (using the homemade ExportXML function) and catching filename
                            sFilesSavedList = sFilesSavedList & vbCrLf & vbTab & "- " & sExportResult

                        End If
                    End If

                End If

                If chkUK.Value Then

                    sExportResult = ExportXML(1, sFileSaveDirectory, 2)

                    If Not sExportResult Like "ERR_*" Then

                        '... Save the file (using the homemade ExportXML function) and catching filename
                        sFilesSavedList = sFilesSavedList & vbCrLf & vbTab & "- " & sExportResult

                    End If
                End If

                If chkUSPP.Value Then

                    sExportResult = ExportXML(1, sFileSaveDirectory, 3)

                    If Not sExportResult Like "ERR_*" Then

                        '... Save the file (using the homemade ExportXML function) and catching filename
                        sFilesSavedList = sFilesSavedList & vbCrLf & vbTab & "- " & sExportResult

                    End If
                End If

                If sFilesSavedList <> vbNullString Then

                    MsgBox "Le(s) fichier(s) XML :" & vbCrLf & sFilesSavedList & vbCrLf & vbCrLf & "a/ont bien été enregistré(s).", vbInformation + vbOKOnly, "Fichier(s) XML bien exporté(s)"

                Else

                    MsgBox "Aucun fichier XML n'a été enregistré.", vbExclamation + vbOKOnly, "Aucun fichier XML exporté"

                End If
            Else 'if no folder was selected...

                '... display a warning message accordingly
                MsgBox "Aucun répertoire choisi. Aucun fichier XML n'a été exporté.", vbExclamation + vbOKOnly, "Pas de répertoire choisi"

            End If
        End With
    Else 'if no checkbox has been selected...

        '... display a warning message accordingly
        MsgBox "Aucun périmètre/programme n'a été choisi." & vbCrLf & "Veuillez en choisir au moins 1.", vbExclamation + vbOKOnly, "Pas de périmètre/programme choisi"

    End If

    Init_Buttons False

End Sub

'Procedure to properly enable/disable all clickable buttons and checkboxes of the Dashboard according to the loading status of some file
'This is inherited from a previous version of the tool where the loading order was forced, enabling the buttons only when the previous expected file was loaded
'Since then, it has been devised that only the Instruments file was mandatory as the 1st load, so the check has been deported
Public Sub Init_Buttons(Optional ByVal bIsStandardExec As Boolean = True)

    Dim iCntSixInstruments, iSixFileNbDefault, iCntSixOptions, iCntSixStructured, iCntUSPPIsins As Integer

    'Counting how many rows of each file are already in the database
    iCntSixInstruments = CurrentDb.OpenRecordset("SELECT COUNT(*) AS CountRows FROM SIX_ISIN WHERE rsf_id = 0")!CountRows
    iCntSixOptions = CurrentDb.OpenRecordset("SELECT COUNT(*) AS CountRows FROM SIX_ISIN WHERE rsf_id = 1")!CountRows
    iCntSixStructured = CurrentDb.OpenRecordset("SELECT COUNT(*) AS CountRows FROM SIX_ISIN WHERE rsf_id = 2")!CountRows
    'commented and reviewed for handling inactive isin (AER_EVOLLM-3)
    'iCntUSPPIsins = CurrentDb.OpenRecordset("SELECT COUNT(*) AS CountRows FROM SIX_ISIN WHERE isin NOT LIKE '" & sInstruNullIsinPattern & "*' AND active_flag <> 'Inactive' AND " & GetUSPPISINSqlFilter & " AND sanctioned = 'Yes' AND confidence_level > 25")!CountRows
    If chkExportInactiveISIN.Value = False Then
        iCntUSPPIsins = CurrentDb.OpenRecordset("SELECT COUNT(*) AS CountRows FROM SIX_ISIN WHERE isin NOT LIKE '" & sInstruNullIsinPattern & "*' AND active_flag <> 'Inactive' AND " & GetUSPPISINSqlFilter & " AND sanctioned = 'Yes' AND confidence_level > 25")!CountRows
    Else
        iCntUSPPIsins = CurrentDb.OpenRecordset("SELECT COUNT(*) AS CountRows FROM SIX_ISIN WHERE isin NOT LIKE '" & sInstruNullIsinPattern & "*' AND " & GetUSPPISINSqlFilter & " AND sanctioned = 'Yes' AND confidence_level > 25")!CountRows
    End If

    'Enabling/disabling Import buttons
    btnImportSixInstruments.Enabled = True
    'Disabled but to reactivate if Options comes back in scope
    btnImportSixOptions.Enabled = False
    btnImportSixStructured.Enabled = True

    'Enable Export button if some rows were found/disable it if not and USPP checkboxes if USPP data material has been found
    btnExportToXML.Enabled = (iCntSixInstruments + iCntSixOptions + iCntSixStructured > 0)
    chkSplitUSPP.Enabled = (btnExportToXML.Enabled And iCntUSPPIsins > 0)
    chkSplitUSPP.Value = IIf(bIsStandardExec, (btnExportToXML.Enabled And iCntUSPPIsins > 0), chkSplitUSPP.Value)
    chkSplitUSPP_Click
    chkEUUS.Enabled = btnExportToXML.Enabled
    chkEUUS_Click
    chkUK.Enabled = btnExportToXML.Enabled
    activateBtnMajApplicationDates

    'Enable/Disable Option panel + Setting default valor
    With PnlOptFormatSIXCSV
        If iCntSixInstruments > 0 Then

            .Value = CurrentDb.OpenRecordset("SELECT DISTINCT rsf_file_nb FROM SIX_ISIN WHERE rsf_id = 0")!rsf_file_nb
            .Enabled = False
            setSixFileInstruDate
            setSixFileInstruTime

        Else

            .Enabled = True

        End If
    End With

    'Enable all other buttons
    btnReset.Enabled = True
    btnSetFilterIsinUSPP.Enabled = True
    btnOriginalUSPPFilter.Enabled = True
    btnFlushAllApplicationDates.Enabled = True
    btnUSPPFilterHelp.Enabled = True
    PnlOptDateFormat.Enabled = True
    btnSetinactiveExclusion.Enabled = True
    btnAutoProcess.Enabled = True

End Sub

            Init_All
            Exit Sub
    End Select

    If DebugStatus Then Debug.Print "Selected file : " & sInstrumentFile
    If DebugStatus Then Debug.Print "Selected file : " & sStructureFile
    If DebugStatus Then Debug.Print "Selected file : " & sApplicationDateFileISIN
    If DebugStatus Then Debug.Print "Selected file : " & sApplicationDateFileUK
    If DebugStatus Then Debug.Print "Selected file : " & sApplicationDateFileCMIC
    If DebugStatus Then Debug.Print "Selected file : " & sApplicationDateFileUSPP

    lStartTickCount = GetTickCount
    lblAutoImport.FontSize = 8
    lblAutoImport.Caption = "Selected Dates files : " & vbCrLf & _
                            " Dates EU/US : " & GetFilenameFromPath(sApplicationDateFileISIN) & vbCrLf & _
                            " Dates UK : " & GetFilenameFromPath(sApplicationDateFileUK) & vbCrLf & _
                            " Dates CMIC : " & GetFilenameFromPath(sApplicationDateFileCMIC) & vbCrLf & _
                            " Dates USPP : " & GetFilenameFromPath(sApplicationDateFileUSPP) & vbCrLf & _
                            "Output Folder :" & vbCrLf & "  " & sGenerationOuputFolder

    lblProgressStep.FontSize = 10

    lblProgressStep.Caption = "Importing SIX Instrument : " & vbCrLf & "   - " & GetFilenameFromPath(sInstrumentFile)
    'Launch ad hoc homemade (Core) procedure to import a Six file. Param 0 --> Instruments file + Option Panel Value for file nb version (Old or New CSV format)
    bFileImported = AutoImportSixFile(0, PnlOptFormatSIXCSV.Value, sInstrumentFile)

    bIsUnwantedInstruRemoved = False

    'Refresh the text displayed for the SIX Instruments loaded file
    RefreshSixInstrumentsLabel

    'Check if an Instruments file has been loaded...
    If CurrentDb.OpenRecordset("SELECT COUNT(*) AS CountRows FROM CURRENTLY_LOADED_FILES WHERE six_instruments_file IS NOT NULL")!CountRows > 0 Then

        If Not bIsUnwantedInstruRemoved Then

            lblProgressStep.Caption = "Importing SIX Structured : " & vbCrLf & "   - " & GetFilenameFromPath(sStructureFile)
            'Launch ad hoc homemade (Core) procedure to import a Six file. Param 1 --> Options file + Option Panel Value for file nb version (Old or New CSV format)
            bFileImported = AutoImportSixFile(2, PnlOptFormatSIXCSV.Value, sStructureFile)

            'Refresh the text displayed for the SIX Options loaded file
            RefreshSixStructuredLabel

        Else

            '... display a warning message accordingly and do nothing
            MsgBox "Un export XML a déjà eu lieu et a tronqué une partie des informations référentielles du fichier ""Instruments"" actuellement chargé (processus normal)." & vbCrLf & "Le fichier ""Options"" a besoin de l'intégralité des données provenant du fichier ""Instruments"" pour être correctement importé.", vbExclamation + vbOKOnly, "Fichier ""Instruments"" partiellement tronqué suite à Export"

        End If

    Else 'if no Instruments file has been previously loaded...

        '... display a warning message accordingly and do nothing
        MsgBox "Aucun fichier ""Instruments"" n'a été chargé." & vbCrLf & "Le fichier ""Options"" a besoin du fichier ""Instruments"" pour être correctement importé.", vbExclamation + vbOKOnly, "Fichier ""Instruments"" manquant"

    End If

    Init_Buttons
    Block_Buttons


    'Launching homemade (Core) import method depending on which lists checkboxes are checked
    If chkEUUS.Value And chkEUUSCMIC.Value Then
        lblProgressStep.Caption = "Importing CMIC Dates : " & vbCrLf & "   - " & GetFilenameFromPath(sApplicationDateFileCMIC)
        AutoImportReglissFile 0, sApplicationDateFileCMIC

    End If


    If chkEUUS.Value And chkEUUSISIN.Value Then
        lblProgressStep.Caption = "Importing EU/US Dates : " & vbCrLf & "   - " & GetFilenameFromPath(sApplicationDateFileISIN)
        AutoImportReglissFile 1, sApplicationDateFileISIN

    End If
    Init_Buttons
    If chkUK.Value Then
        lblProgressStep.Caption = "Importing UK Dates : " & vbCrLf & "   - " & GetFilenameFromPath(sApplicationDateFileUK)
        AutoImportReglissFile 2, sApplicationDateFileUK

    End If

    If chkUSPP.Value Then
        lblProgressStep.Caption = "Importing USPP Dates : " & vbCrLf & "   - " & GetFilenameFromPath(sApplicationDateFileUSPP)
        AutoImportReglissFile 3, sApplicationDateFileUSPP

    End If

    'Generating files
    lblProgressStep.Caption = "Exporting File : "
    Dim sFileSaveDirectory, sFilesSavedList, sExportResult As String

    If (chkEUUS.Value And (chkEUUSCMIC.Value Or chkEUUSISIN.Value)) Or chkUK.Value Or chkUSPP.Value Then

        'If unwated Instruments have not yes been removed...
        If Not bIsUnwantedInstruRemoved Then

            '...calling the homemade procedure which ultimately removes the unwanted Instruments that were kept for proper integration of the Options/Structured data
            RemoveUnwantedInstruments
            bIsUnwantedInstruRemoved = True

        End If

        'Calling the filler for ISIN PROGRAMS table now that only the ISIN we want are present in the main table
        FillIsinPrograms

            '... save the path
            sFileSaveDirectory = sGenerationOuputFolder
            sFilesSavedList = vbNullString
            lblProgressStep.Caption = "Exporting Files to : " & vbCrLf & "   - " & sFileSaveDirectory

            If chkEUUS.Value Then

                If chkEUUSCMIC.Value Then
                lblProgressStep.Caption = "Exporting CMIC File"
                    '... Save the file (using the homemade ExportXML function) and catching filename

                    sExportResult = ExportXML(0, sFileSaveDirectory, 1)

                    If Not sExportResult Like "ERR_*" Then

                        sFilesSavedList = sFilesSavedList & vbCrLf & vbTab & "- " & sExportResult

                    End If
                End If

                If chkEUUSISIN.Value Then
                    lblProgressStep.Caption = "Exporting EU/US File"
                    sExportResult = ExportXML(1, sFileSaveDirectory, 1)

                    If Not sExportResult Like "ERR_*" Then

                        '... Save the file (using the homemade ExportXML function) and catching filename
                        sFilesSavedList = sFilesSavedList & vbCrLf & vbTab & "- " & sExportResult

                    End If
                End If

            End If

            If chkUK.Value Then
                lblProgressStep.Caption = "Exporting UK File"
                sExportResult = ExportXML(1, sFileSaveDirectory, 2)

                If Not sExportResult Like "ERR_*" Then

                    '... Save the file (using the homemade ExportXML function) and catching filename
                    sFilesSavedList = sFilesSavedList & vbCrLf & vbTab & "- " & sExportResult

                End If
            End If

            If chkUSPP.Value Then
                lblProgressStep.Caption = "Exporting USPP File"
                sExportResult = ExportXML(1, sFileSaveDirectory, 3)

                If Not sExportResult Like "ERR_*" Then

                    '... Save the file (using the homemade ExportXML function) and catching filename
                    sFilesSavedList = sFilesSavedList & vbCrLf & vbTab & "- " & sExportResult

                End If
            End If

            If sFilesSavedList <> vbNullString Then

                'MsgBox "Le(s) fichier(s) XML :" & vbCrLf & sFilesSavedList & vbCrLf & vbCrLf & "a/ont bien été enregistré(s).", vbInformation + vbOKOnly, "Fichier(s) XML bien exporté(s)"
                Debug.Print "Le(s) fichier(s) XML :" & vbCrLf & sFilesSavedList & vbCrLf & vbCrLf & "a/ont bien été enregistré(s)."
                lStopTickCount = GetTickCount

                lTimerValue = ((lStopTickCount - lStartTickCount) + Abs((lStopTickCount < lStartTickCount) * 86400)) / 1000
                Select Case lTimerValue
                    Case Is < 1
                        sTimerMsg = Format(lTimerValue, "0.000 \m\s")
                    Case 1 To 59
                        sTimerMsg = Format(Int(lTimerValue), "0 \s") 'this one rounds down
                        'mssg = mssg & Format(timerTimeValue\1, "0 \s") this one rounds it off up or down
                    Case 60 To 3599
                        sTimerMsg = Format(Int(lTimerValue / 60), "0 \m\, ") & Format(lTimerValue Mod 60, "0 \s")
                    Case Is >= 3600
                        sTimerMsg = Format(Int(lTimerValue / 3600), "0 \h\, ") & Format(Int((lTimerValue Mod 3600) / 60), "0 \m\, ") & Format(lTimerValue Mod 60, "0 \s")
                    Case Else
                        'do nothing
                End Select

                lblProgressStep.Caption = "Total Time to process : " & vbCrLf & "   - " & sTimerMsg

            Else

                MsgBox "Aucun fichier XML n'a été enregistré.", vbExclamation + vbOKOnly, "Aucun fichier XML exporté"

            End If
        Else 'if no folder was selected...

            '... display a warning message accordingly
            MsgBox "Aucun répertoire choisi. Aucun fichier XML n'a été exporté.", vbExclamation + vbOKOnly, "Pas de répertoire choisi"

        End If

    lblProgressStep.Caption = "Total Time to process : " & vbCrLf & "   - " & sTimerMsg
    'Re-enabling interface
    Init_Buttons False

End Sub

'Actions taken when closing the Dashboard form (hence, when quitting Access/closing the Access file too)
Private Sub Form_Close()

    'Flush all the SIX data from the database with homemade (Core) procedure
    If CurrentDb.OpenRecordset("SELECT COUNT(*) AS CountRows FROM SIX_ISIN")!CountRows > 0 Then

        If MsgBox("Voulez-vous conservez les données SIX Financial déjà chargées ?", vbQuestion + vbYesNo, "Conserver données SIX Financial ?") = vbNo Then

            FlushAll

        Else

            FlushTempDocsFromHDD

        End If
    Else

        FlushTempDocsFromHDD

    End If

    'If some Application Dates have been loaded...
    If CurrentDb.OpenRecordset("SELECT COUNT(*) AS CountRows FROM ISIN_HISTO_APPLICATION_DATE")!CountRows > 0 Then

        '... asking if they should be kept...
        If MsgBox("Voulez-vous conserver les Application Dates déjà chargées ?", vbQuestion + vbYesNo, "Conserver Application Dates ?") = vbNo Then

            '... if no, then flush them with homemade (Core) procedure
            FlushISINHistoApplicationDate

        End If
    End If
End Sub

'Actions taken when opening the Dashboard form (hence, when opening the Acces file too)
Private Sub Form_Open(Cancel As Integer)

    'Initialize Dashboard display
    Init_All

End Sub
